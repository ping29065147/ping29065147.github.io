# 思路

以變數i控制儲存位置，直接覆蓋重複的值

# 參考程式碼

class Solution {
public:
    int removeDuplicates(vector<int>& nums)
    {
        int i = 0;
        for (int j = 0; j < nums.size(); ++j)
           if (nums[i] != nums[j])
               nums[++i] = nums[j];
        
        return i + 1;
    }
};
